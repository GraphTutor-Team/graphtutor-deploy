## 📝 使用示例

### 基础事务使用

```javascript
import { withTransaction } from '../utils/transaction.js';

const result = await withTransaction(async (session) => {
  const question = await Question.create([data], { session });
  await QuestionBank.findByIdAndUpdate(bankId, 
    { $inc: { questionCount: 1 } }, 
    { session }
  );
  return question[0];
});
```

### 业务操作使用

```javascript
import { createQuestionWithBankUpdate } from '../utils/businessOperations.js';

const question = await createQuestionWithBankUpdate(
  questionData,
  userId
);
```

### 一致性检查使用

```javascript
import { checkAllConsistency, fixConsistencyIssues } from '../utils/dataConsistency.js';

const result = await checkAllConsistency();
if (!result.isConsistent) {
  await fixConsistencyIssues(result, { autoFix: true });
}
```
