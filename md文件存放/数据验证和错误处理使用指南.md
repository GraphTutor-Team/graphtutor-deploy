# 数据验证和错误处理使用指南

**创建时间**: 2026-01-19

## 📋 概述

本项目已实现统一的数据验证和错误处理系统，提供：
- 统一的验证规则和验证器
- 友好的错误响应格式
- 自动错误处理中间件
- 类型安全的错误类

## 🎯 核心功能

### 1. 验证器系统

#### 验证规则（ValidationRules）

```javascript
import { ValidationRules } from '../utils/validator.js';

// 字符串验证
ValidationRules.string.required('此字段为必填项')
ValidationRules.string.minLength(5, '至少5个字符')
ValidationRules.string.maxLength(100, '最多100个字符')
ValidationRules.string.pattern(/^\S+$/, '不能包含空格')

// 数字验证
ValidationRules.number.required('此字段为必填项')
ValidationRules.number.min(1, '必须大于1')
ValidationRules.number.max(100, '必须小于100')
ValidationRules.number.range(1, 100, '必须在1-100之间')
ValidationRules.number.integer('必须是整数')

// 数组验证
ValidationRules.array.required('此字段为必填项')
ValidationRules.array.minLength(2, '至少2项')
ValidationRules.array.maxLength(10, '最多10项')

// 其他验证
ValidationRules.email('邮箱格式不正确')
ValidationRules.url('URL格式不正确')
ValidationRules.objectId('无效的ID格式')
ValidationRules.enum(['value1', 'value2'], '值必须是value1或value2')
```

#### 常用验证模式（CommonSchemas）

```javascript

// 邮箱验证

// 密码验证

// 难度验证

```

### 3. 错误处理

#### 错误类

```javascript
import {
  AppError,
  ValidationError,
  UnauthorizedError,
  ForbiddenError,
  NotFoundError,
  ConflictError,
} from '../utils/errorHandler.js';

// 抛出验证错误
throw new ValidationError('验证失败', ['字段1错误', '字段2错误']);

// 抛出未授权错误
throw new UnauthorizedError('请先登录');

// 抛出权限不足错误
throw new ForbiddenError('您没有权限执行此操作');

// 抛出未找到错误
throw new NotFoundError('题目不存在');

// 抛出冲突错误
throw new ConflictError('邮箱已被注册');

// 自定义错误
throw new AppError('自定义错误消息', 400, 'CUSTOM_ERROR', { detail: '详细信息' });
```

#### 异步错误处理

```javascript
import { asyncHandler } from '../utils/errorHandler.js';

// 使用asyncHandler包装异步函数，自动捕获错误
router.get('/api/questions/:id',
  asyncHandler(async (req, res) => {
    const question = await Question.findById(req.params.id);
    if (!question) {
      throw new NotFoundError('题目不存在');
    }
    res.json({ success: true, data: question });
  })
);
```

#### 成功响应

```javascript
import { successResponse, paginatedResponse } from '../utils/errorHandler.js';

// 简单成功响应
res.json(successResponse(data, '操作成功', 200));

// 分页响应
res.json(paginatedResponse(items, page, limit, total, '获取成功'));
```

## 📝 使用示例

### 示例1: 创建题目API

```javascript
import express from 'express';
import { validateBody, CommonSchemas, ValidationRules } from '../middleware/validateRequest.js';
import { asyncHandler, successResponse, NotFoundError } from '../utils/errorHandler.js';
import { requireAuth } from '../middleware/auth.js';
import Question from '../models/Question.js';

const router = express.Router();

router.post('/',
  requireAuth,
  validateBody({
    content: CommonSchemas.questionContent,
    options: CommonSchemas.questionOptions,
    answerIndex: [
      ValidationRules.number.required('答案索引不能为空'),
      ValidationRules.number.integer('答案索引必须是整数'),
      ValidationRules.number.min(0, '答案索引不能小于0'),
    ],
    difficulty: CommonSchemas.difficulty,
    explain: [
      ValidationRules.string.maxLength(2000, '解析不能超过2000个字符'),
    ],
  }),
  asyncHandler(async (req, res) => {
    const { content, options, answerIndex, difficulty, explain } = req.body;
    
    // 验证答案索引是否在有效范围内
    if (answerIndex >= options.length) {
      throw new ValidationError('答案索引超出范围', [
        `答案索引必须在0-${options.length - 1}之间`,
      ]);
    }

    const question = await Question.create({
      content,
      options,
      answerIndex,
      difficulty,
      explain,
      createdBy: req.user.id,
    });

    res.status(201).json(successResponse(question, '题目创建成功', 201));
  })
);

export default router;
```

### 示例2: 获取题目列表API

```javascript
import { validateQuery, CommonSchemas, ValidationRules } from '../middleware/validateRequest.js';
import { asyncHandler, paginatedResponse } from '../utils/errorHandler.js';

router.get('/',
  requireAuth,
  validateQuery({
    page: CommonSchemas.page,
    limit: CommonSchemas.limit,
    difficulty: [
      ValidationRules.number.range(1, 5, '难度必须在1-5之间'),
    ],
    search: [
      ValidationRules.string.maxLength(100, '搜索关键词不能超过100个字符'),
    ],
  }),
  asyncHandler(async (req, res) => {
    const { page = 1, limit = 20, difficulty, search } = req.query;
    
    const query = {};
    if (difficulty) query.difficulty = difficulty;
    if (search) query.content = { $regex: search, $options: 'i' };

    const skip = (page - 1) * limit;
    const [items, total] = await Promise.all([
      Question.find(query).skip(skip).limit(limit),
      Question.countDocuments(query),
    ]);

    res.json(paginatedResponse(items, page, limit, total));
  })
);
```

### 示例3: 更新题目API

```javascript
router.put('/:id',
  requireAuth,
  validate({
    params: {
      id: [ValidationRules.string.required(), ValidationRules.objectId()],
    },
    body: {
      content: [
        ValidationRules.string.maxLength(2000, '题干不能超过2000个字符'),
      ],
      difficulty: CommonSchemas.difficulty,
    },
  }),
  asyncHandler(async (req, res) => {
    const { id } = req.params;
    const question = await Question.findById(id);
    
    if (!question) {
      throw new NotFoundError('题目不存在');
    }

    // 检查权限
    if (question.createdBy !== req.user.id && !req.user.isAdmin) {
      throw new ForbiddenError('您没有权限修改此题目');
    }

    Object.assign(question, req.body);
    await question.save();

    res.json(successResponse(question, '题目更新成功'));
  })
);
```

## 📚 最佳实践

### 1. 验证顺序
- 先验证路径参数
- 再验证查询参数
- 最后验证请求体

### 2. 错误处理
- 使用 `asyncHandler` 包装所有异步路由处理函数
- 使用适当的错误类抛出错误
- 提供清晰的错误消息

### 3. 验证规则
- 优先使用 `CommonSchemas` 中的预定义模式
- 自定义验证规则要清晰明确
- 提供友好的错误消息

### 4. 响应格式
- 使用 `successResponse` 和 `paginatedResponse` 保持一致性
- 错误响应会自动格式化，无需手动处理

## 🚀 迁移指南

### 从旧代码迁移

**旧代码**:
```javascript
router.post('/api/questions', async (req, res) => {
  try {
    const { content, options } = req.body;
    
    if (!content) {
      return res.status(400).json({ error: '题干不能为空' });
    }
    
    if (!Array.isArray(options) || options.length < 2) {
      return res.status(400).json({ error: '至少需要2个选项' });
    }
    
    const question = await Question.create({ content, options });
    res.json(question);
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: '创建失败' });
  }
});
```

**新代码**:
```javascript
import { validateBody, CommonSchemas } from '../middleware/validateRequest.js';
import { asyncHandler, successResponse } from '../utils/errorHandler.js';

router.post('/api/questions',
  validateBody({
    content: CommonSchemas.questionContent,
    options: CommonSchemas.questionOptions,
  }),
  asyncHandler(async (req, res) => {
    const { content, options } = req.body;
    const question = await Question.create({ content, options });
    res.status(201).json(successResponse(question, '创建成功', 201));
  })
);
```
